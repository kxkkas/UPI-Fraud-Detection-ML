create database fraud;
use fraud;

create table register(username varchar(50) primary key,
password varchar(50),
contact varchar(15),
email varchar(50),
address varchar(80));

CREATE TABLE blocked_accounts (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nameOrig VARCHAR(100) UNIQUE,
    reason VARCHAR(255),
    blocked_on TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE TABLE prediction_history (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50),
    nameOrig VARCHAR(100),
    tx_type VARCHAR(20),
    amount DOUBLE,
    confidence DOUBLE,
    result VARCHAR(50),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
